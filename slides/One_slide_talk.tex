\begin{column}{0.48\textwidth}

  \textbf{Operator learning problem:}
\[
  \mathcal{G}:\mathcal{U}\times\mathcal{K}
  \to L^2(\Omega\times\mathcal{T}),
  \quad (u,\kappa)\mapsto s
\]
{\footnotesize
Learn $\hat{\mathcal{G}} \approx \mathcal{G}$ from data $\mathcal{D}$
without solving the PDE.
}

  \vspace{0.1cm}

  \begin{block}{Challenge}
  \small
  A single basis fails when solutions
  change character fundamentally across~$\kappa$.
  \end{block}

  \vspace{0.1cm}

 \begin{block}{TEMPO}
\small
Discover $M$ latent regimes via EM-GMM,
build a dedicated basis $\{\Phi_m\}$ per regime
\textit{(POD or NeuralPOD)},
route new inputs at inference time.
\end{block}

  \vspace{0.1cm}

  \footnotesize
  \begin{tabular}{ll}
    $\Phi_k^{(m)}$ & POD mode $k$ of regime $m$ \\[2pt]
    $w_m(u,\kappa)$ & gating weight \\[2pt]
    $\gamma_{im}$ & soft EM assignment \\
  \end{tabular}
