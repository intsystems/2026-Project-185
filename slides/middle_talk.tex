\begin{frame}{Problem Statement}

\begin{block}{Task}
  Given $\cD = \{(u_0^{(i)},\kappa^{(i)},s^{(i)})\}_{i=1}^N$,
  construct $\hat{\cG}\approx\cG\colon(u_0,\kappa)\mapsto s$
  accurate across all qualitative regimes, without solving the PDE.
\end{block}

\begin{block}{Problem}
  Existing methods learn a single basis for all $\kappa$.
  When solutions change character fundamentally across parameter values —
  shock structures at $\nu{=}0.001$ versus smooth diffusion at $\nu{=}1.0$ —
  the basis must compromise between incompatible geometries,
  concentrating error precisely where accuracy matters most.
\end{block}


\medskip
\textbf{Proposal:} discover latent regimes unsupervised and build a dedicated basis per regime —{\textbf{TEMPO}}

\end{frame}





\begin{frame}{TEMPO: Central Idea — Regime-Aware Operator Learning}

\begin{columns}[T]

\begin{column}{0.48\textwidth}
  \centering
  \includegraphics[width=0.82\textwidth]{TEMPO/main/pictures/TEMPO_snaps.png}

  \vspace{0.12cm}
  \small
  \textbf{Offline:} EM-GMM $\Rightarrow$ $M$ bases $\{\Phi_m\}$\\[4pt]
  \textbf{Online:}
  $\displaystyle\hat{s} = \sum_m w_m\Bigl[\phi_0^{(m)} + \sum_k b_{mk}\,\phi_k^{(m)}\Bigr]$


\end{column}

\begin{column}{0.48\textwidth}

  \[
    \cG:\cU\times\cK \to L^2(\Omega\times\cT),
    \quad (u_0,\kappa)\mapsto s
  \]
  {\footnotesize Learn $\hat{\cG}\approx\cG$ from $\cD$ without solving the PDE.}

  \vspace{0.2cm}

  \begin{block}{TEMPO}
  \small
  Discover $M$ latent regimes via EM-GMM.
  Build a dedicated basis per regime \textit{(POD or NeuralPOD)}.
  Route new inputs at inference with a learned gating network.
  \end{block}

  \vspace{0.15cm}



\end{column}
\end{columns}

\end{frame}






\begin{frame}{Related Work}

\begin{itemize}
  \setlength\itemsep{0.55em}

  \textbf{DeepONet} \hfill {\footnotesize Lu et al., 2019}\\
    {\small Branch encodes input function; trunk encodes query location.
    Universal approximation of nonlinear operators.}

   \textbf{POD-DeepONet} \hfill {\footnotesize Lu et al., 2022}\\
    {\small Fixed POD basis replaces the trunk network.
    Compact, interpretable, data-efficient.}

   \textbf{Neural-POD} \hfill {\footnotesize Mou et al., 2026}\\
    {\small Nonlinear orthogonal basis via greedy residual minimization.
    Captures structure beyond any linear subspace.}

   \textbf{Local Reduced-Order Bases} \hfill {\footnotesize Amsallem et al., 2012}\\
    {\small Snapshot space partitioned into clusters; separate POD basis
    per cluster. Hard assignment, offline only.}

   \textbf{EM Algorithm} \hfill {\footnotesize Dempster, Laird, Rubin, 1977}\\
    {\small Maximum likelihood with latent variables.
    Monotone log-likelihood convergence at every iteration.}

\end{itemize}

\vspace{0.1cm}
{\footnotesize\textit{TEMPO combines Neural-POD bases with soft EM regime discovery,
integrating local basis construction and operator learning end-to-end.}}

\end{frame}







\begin{frame}{Problem Statement}

\textbf{Hypothesis.} Each trajectory arises from one of $M$ latent regimes; labels $z_i$ are unobserved.

\medskip

\begin{columns}[c]
\begin{column}{0.54\textwidth}
  \small
  \begin{align*}
    z_i
      &\sim \mathrm{Categorical}(\pi_1,\ldots,\pi_M) \\[5pt]
    \bigl(s^{(i)} \mid z_i{=}m\bigr)
      &\sim \mathcal{N}\!\bigl(\hat{s}_m^{(i)},\;\sigma^2 W^{-1}\bigr) \\[5pt]
    \hat{s}_m^{(i)}(y)
      &= \phi_0^{(m)}(y)
         + \sum_{p=1}^{P_m}\lambda_p^{(m,i)}\,\phi_p^{(m)}(y)
  \end{align*}
\end{column}
\begin{column}{0.42\textwidth}
  \small\renewcommand{\arraystretch}{1.5}
  \begin{tabular}{ll}
    $z_i$               & regime label (latent)  \\
    $\pi_m$             & prior weight           \\
    $\sigma^2$          & assignment softness    \\
    $\phi_p^{(m)}$      & spatiotemporal mode    \\
    $\lambda_p^{(m,i)}$ & scalar amplitude       \\
  \end{tabular}
\end{column}
\end{columns}

\medskip

\begin{columns}[c]
\begin{column}{0.54\textwidth}
  \small Find $\Phi^*,\pi^*,\sigma^{2*}$
  maximising the observed-data log-likelihood~(EM).
\end{column}
\begin{column}{0.42\textwidth}
  \[
    \varepsilon = \frac{1}{N_{\mathrm{test}}}
    \sum_{i}\frac{\|s^{(i)}-\hat{s}^{(i)}\|}{\|s^{(i)}\|}
  \]
\end{column}
\end{columns}

\end{frame}






\begin{frame}{Solution}

\begin{center}\small
  \textbf{POD}~(linear SVD)
    $\;\to\;$ POD-DeepONet ~|~ \textcolor{red!70}{\textbf{TEMPO(POD)}}\\[3pt]
  \textbf{NeuralPOD}~(greedy MLP)
    $\;\to\;$ NeuralPOD-DeepONet ~|~ \textcolor{red!70}{\textbf{TEMPO(NeuralPOD)}}
\end{center}

\smallskip

\begin{columns}[T]
\begin{column}{0.49\textwidth}
  \small\textbf{Phase 1 — Offline: obtain $\boldsymbol{\lambda}^{(i)}$}\\[4pt]

  \textit{POD:}\enspace
  $S \approx U_K\Sigma_K V_K^\top$,\quad
  $\lambda_k^{(i)} = \langle s^{(i)}, \phi_k\rangle$

  \smallskip
  \textit{NeuralPOD:}
  \[
    \min_{\phi_k,\,\{\lambda_k^{(i)}\}}
    \frac{1}{N}\sum_{i=1}^N
    \bigl\|r^{(k-1,i)} - \lambda_k^{(i)}\phi_k\bigr\|^2
  \]

  \textit{TEMPO:} same, but $\gamma_{im}$-weighted. For POD:
  \[
    \bar{s}_m = \textstyle\sum_i \gamma_{im} s^{(i)},\quad
    S_w = \mathrm{diag}(\sqrt{\gamma_{\cdot m}})\,\tilde{S}
  \]
\end{column}
\begin{column}{0.49\textwidth}
  \small\textbf{Phase 2 — Online: train BranchNet on $\boldsymbol{\lambda}^{(i)}$}\\[4pt]

  POD-DeepONet / NeuralPOD-DeepONet:
  \[
    \min_\theta\;\frac{1}{N}\sum_{i=1}^N
    \bigl\|\mathcal{B}_\theta(u_0^{(i)},\kappa^{(i)}) - \boldsymbol{\lambda}^{(i)}\bigr\|^2
  \]

  TEMPO (GatingNet $+$ $M$ BranchNets):
  \[
    \mathcal{L}_{\text{online}}
    = \mathcal{L}_{\text{data}}
    + \lambda_{\text{KL}}\,\mathcal{L}_{\text{KL}}
    - \lambda_H\,\mathcal{L}_{\text{ent}}
  \]
\end{column}
\end{columns}

\vspace{0.15cm}
{\footnotesize\itshape
  The offline $\boldsymbol{\lambda}^{(i)}$ are the regression targets for the online
  BranchNet in all four methods.}

\end{frame}








\begin{frame}{Computational Experiment}

\begin{block}{Dataset: 1D Burgers' equation (PDEBench)}
  \[
    u_t + u\,u_x = \nu\,u_{xx},\quad x\in(-1,1),\quad t\in(0,2]
  \]
  \small
  $\nu\in\{0.001,\,0.01,\,0.1,\,1.0\}$ — three orders of magnitude.
  $\;9{,}500$ trajectories per $\nu$ $\;\Rightarrow\;$ $38{,}000$ total.
  Grid: $N_x{=}1{,}024$, $N_t{=}101$.
\end{block}

\medskip

\begin{columns}[T]
\begin{column}{0.48\textwidth}
  \small\textbf{Three verification claims:}
  \begin{enumerate}[1)]
    \item Single-regime bases concentrate error at regime extremes
    \item EM regimes cross $\nu$ reflects $\kappa$ and geometry
    \item TEMPO outperforms all baselines across the full $\nu$ range
  \end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
  \small\textbf{Methods compared:}\\[4pt]
  \begin{tabular}{ll}
    DeepONet           & baseline \\
    POD-DeepONet       & linear, 1 regime \\
    NeuralPOD-DeepONet & nonlinear, 1 regime \\
    \textcolor{red!70}{\textbf{TEMPO(POD)}}       & linear, $M$ regimes \\
    \textcolor{red!70}{\textbf{TEMPO(NeuralPOD)}} & nonlinear, $M$ regimes \\
  \end{tabular}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Conclusion}

\begin{block}{Contributions}
  \begin{itemize}
    \item TEMPO: offline EM-GMM regime discovery $+$ dedicated basis per regime $+$ gated online operator
    \item NeuralPOD-DeepONet as a strong nonlinear single-regime baseline
    \item EM regimes reflect solution geometry and $\kappa$ labels
  \end{itemize}
\end{block}

\begin{block}{Main result}
  Regime-specific bases achieve lower reconstruction error at similar mode count;
  the gated operator outperforms all single-basis baselines
\end{block}

\begin{block}{Future work}
  \begin{itemize}
    \item Parameter-conditioned modes $\phi_k^{(m)}(\cdot;\kappa)$
    \item Extension to 2D (Darcy flow)
    \item Convergence theory for the GEM update
  \end{itemize}
\end{block}

\end{frame}
